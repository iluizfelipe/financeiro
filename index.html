🔗 Sincronização com Google Sheets

Passo 1 — Acesse script.google.com → Novo projeto → apague tudo e cole o código abaixo:
const SHEET_NAME = 'FinanceApp'; function doGet(e) { return handle(e); } function doPost(e) { return handle(e); } function handle(e) { const cors = {'Access-Control-Allow-Origin':'*'}; try { const id = e.parameter.sheetId; const ss = SpreadsheetApp.openById(id); let sh = ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME); const m = e.parameter.method || 'read'; if (m === 'ping') return out(JSON.stringify({ok:true})); if (m === 'read') return out(sh.getRange(1,1).getValue() || '[]'); if (m === 'write') { const body = e.postData ? e.postData.contents : e.parameter.data; sh.getRange(1,1).setValue(body); sh.getRange(1,2).setValue(new Date().toISOString()); return out(JSON.stringify({ok:true})); } } catch(err) { return out(JSON.stringify({error:err.toString()})); } } function out(s) { return ContentService.createTextOutput(s).setMimeType(ContentService.MimeType.JSON); }
